bool CTradingAssistantFixed::Init(void)
{
    Print("=== INITIALIZING APEIRON EDGE v56 ===");
    
    if(m_initialized)
    {
        Print("Already initialized v56");
        return true;
    }
    
    // Clean up any existing objects first
    Deinit();
    
    // Initialize trading object
    m_trade.SetExpertMagicNumber(InpMagicNumber);
    m_trade.SetDeviationInPoints(InpSlippage);
    m_trade.SetTypeFilling(ORDER_FILLING_FOK);
    
    // Set initial values
    m_currentSymbol = Symbol();
    m_autoCalculate = InpAutoCalculate;
    m_isPercentMode = true;
    m_currentTab = 0;
    m_timeDisplayMode = 0;
    m_calculatedLots = 0.01;
    
    // Reset GUI states
    m_dropdownVisible = false;
    m_browserVisible = false;
    m_dropdownOffset = 0;
    m_browserOffset = 0;
    
    // Initialize symbol management
    m_watchlistCount = 0;
    m_totalSymbols = 0;
    m_filteredCount = 0;
    
    // Add current symbol to watchlist
    AddSymbolToWatchlist(m_currentSymbol);
    
    // Create main interface
    CreateMainInterface();
    
    // Initial calculations
    CalculateLotSize();
    
    m_initialized = true;
    m_lastUpdate = TimeCurrent();
    
    Print("✅ APEIRON EDGE v56 INITIALIZED");
    return true;
}

//+------------------------------------------------------------------+
//| Time Display Update - CLEANED                                   |
//+------------------------------------------------------------------+
void CTradingAssistantFixed::UpdateTime(void)
{
    datetime currentTime = TimeCurrent();
    MqlDateTime dt;
    TimeToStruct(currentTime, dt);
    
    string timeStr;
    color btnColor = C'0,0,255'; // Blue
    
    switch(m_timeDisplayMode)
    {
        case 0: // 24-hour format
            timeStr = StringFormat("%02d:%02d:%02d", dt.hour, dt.min, dt.sec);
            btnColor = C'0,0,255'; // Blue
            break;
            
        case 1: // 12-hour format with AM/PM
            {
                int hour12 = dt.hour;
                string ampm = "AM";
                if(hour12 >= 12)
                {
                    ampm = "PM";
                    if(hour12 > 12) hour12 -= 12;
                }
                if(hour12 == 0) hour12 = 12;
                
                timeStr = StringFormat("%d:%02d:%02d %s", hour12, dt.min, dt.sec, ampm);
                btnColor = C'0,128,0'; // Green
            }
            break;
            
        case 2: // Countdown to next candle
            {
                int periodSeconds = PeriodSeconds();
                int secondsElapsed = (int)currentTime % periodSeconds;
                int secondsRemaining = periodSeconds - secondsElapsed;
                
                int hours = secondsRemaining / 3600;
                int minutes = (secondsRemaining % 3600) / 60;
                int seconds = secondsRemaining % 60;
                
                if(hours > 0)
                    timeStr = StringFormat("-%d:%02d:%02d", hours, minutes, seconds);
                else
                    timeStr = StringFormat("-%d:%02d", minutes, seconds);
                    
                btnColor = C'255,165,0'; // Orange
            }
            break;
    }
    
    ObjectSetString(0, "btnTimeToggle_v56_Text", OBJPROP_TEXT, timeStr);
    ObjectSetInteger(0, "btnTimeToggle_v56", OBJPROP_BGCOLOR, btnColor);
    ChartRedraw(0);
}

double CTradingAssistantFixed::CalculatePositionSize(double riskAmount, double stopLossPips)
{
   if(stopLossPips <= 0) return 0.01;
   
   string symbol = m_currentSymbol;
   double point = SymbolInfoDouble(symbol, SYMBOL_POINT);
   double tickValue = SymbolInfoDouble(symbol, SYMBOL_TRADE_TICK_VALUE);
   double tickSize = SymbolInfoDouble(symbol, SYMBOL_TRADE_TICK_SIZE);
   
   if(tickValue > 0 && point > 0)
   {
      double slPoints = stopLossPips * point;
      double valuePerLot = tickValue * (slPoints / tickSize);
      double lots = riskAmount / valuePerLot;
      
      // Apply symbol constraints
      double lotStep = SymbolInfoDouble(symbol, SYMBOL_VOLUME_STEP);
      double minLot = SymbolInfoDouble(symbol, SYMBOL_VOLUME_MIN);
      double maxLot = SymbolInfoDouble(symbol, SYMBOL_VOLUME_MAX);
      
      if(lotStep > 0) lots = MathRound(lots / lotStep) * lotStep;
      if(lots < minLot) lots = minLot;
      if(lots > maxLot) lots = maxLot;
      
      return lots;
   }
   
   return 0.01; // Fallback
}

void CTradingAssistantFixed::CalculateLotSizeWithOverlayIntegration(void)
{
   // Get risk value from chart object
   string riskText = ObjectGetString(0, "edRisk_v56_Text", OBJPROP_TEXT);
   double riskValue = StringToDouble(riskText);
   
   if(riskValue <= 0)
   {
      m_calculatedLots = 0.01;
      ObjectSetString(0, "lblLotSize_v56", OBJPROP_TEXT, "Lots: 0.01 (Risk $0.00)");
      ObjectSetInteger(0, "lblLotSize_v56", OBJPROP_COLOR, C'255,0,0'); // Red
      ChartRedraw(0);
      return;
   }
   
   double balance = AccountInfoDouble(ACCOUNT_BALANCE);
   double riskAmount = 0;
   
   if(!m_autoCalculate)
   {
      // Manual mode - use risk value as lot size directly
      m_calculatedLots = riskValue;
      
      // FIXED: Apply lot size constraints in manual mode too
      string symbol = m_currentSymbol;
      double lotStep = SymbolInfoDouble(symbol, SYMBOL_VOLUME_STEP);
      double minLot = SymbolInfoDouble(symbol, SYMBOL_VOLUME_MIN);
      double maxLot = SymbolInfoDouble(symbol, SYMBOL_VOLUME_MAX);
      
      if(lotStep > 0) m_calculatedLots = NormalizeDouble(MathRound(m_calculatedLots / lotStep) * lotStep, 2);
      if(m_calculatedLots < minLot) m_calculatedLots = minLot;
      if(m_calculatedLots > maxLot) m_calculatedLots = maxLot;
      
      ObjectSetString(0, "lblLotSize_v56", OBJPROP_TEXT, StringFormat("Lots: %.2f (Manual)", m_calculatedLots));
      ObjectSetInteger(0, "lblLotSize_v56", OBJPROP_COLOR, C'0,0,255'); // Blue
      ChartRedraw(0);
      return;
   }
   
   // Calculate risk amount in dollars
   if(m_isPercentMode)
   {
      riskAmount = balance * (riskValue / 100.0);
   }
   else
   {
      riskAmount = riskValue; // Fixed dollar amount
   }
   
   // CRITICAL FIX: Get prices from ACTUAL LINES using NEW multi-overlay system
   CTradingDialog* active_overlay = GetActiveOverlay();
   if(active_overlay == NULL) 
   {
      CalculateLotSizeBasic();
      return;
   }
   
   double main_price = GetLinePriceFromObject(active_overlay.m_main_panel.m_line_name);
   double sl_price = GetLinePriceFromObject(active_overlay.m_sl_panel.m_line_name);
   
   // Fallback to panel prices if line prices are not available
   if(main_price == 0.0)
   {
      main_price = active_overlay.m_main_panel.GetPriceLevel();
   }
   if(sl_price == 0.0)
   {
      sl_price = active_overlay.m_sl_panel.GetPriceLevel();
   }
   
   double sl_distance_points = MathAbs(main_price - sl_price);
   
   // Get symbol information
   string symbol = m_currentSymbol;
   double point = SymbolInfoDouble(symbol, SYMBOL_POINT);
   double tickValue = SymbolInfoDouble(symbol, SYMBOL_TRADE_TICK_VALUE);
   double tickSize = SymbolInfoDouble(symbol, SYMBOL_TRADE_TICK_SIZE);
   
   if(tickValue > 0 && point > 0 && sl_distance_points > 0)
   {
      // Calculate lot size based on actual SL distance from overlay LINES
      double valuePerLot = tickValue * (sl_distance_points / tickSize);
      m_calculatedLots = riskAmount / valuePerLot;
      
      // CRITICAL FIX: Apply symbol constraints with PROPER PRECISION
      double lotStep = SymbolInfoDouble(symbol, SYMBOL_VOLUME_STEP);
      double minLot = SymbolInfoDouble(symbol, SYMBOL_VOLUME_MIN);
      double maxLot = SymbolInfoDouble(symbol, SYMBOL_VOLUME_MAX);
      
      // FIXED: Normalize to 2 decimal places to prevent floating point errors
      if(lotStep > 0) 
      {
         m_calculatedLots = NormalizeDouble(MathRound(m_calculatedLots / lotStep) * lotStep, 2);
      }
      else
      {
         m_calculatedLots = NormalizeDouble(m_calculatedLots, 2);
      }
      
      if(m_calculatedLots < minLot) m_calculatedLots = minLot;
      if(m_calculatedLots > maxLot) m_calculatedLots = maxLot;
      
      // Calculate SL distance in pips for display
      double sl_pips = sl_distance_points / point / 10.0; // Convert to pips
      
      // ENHANCED DISPLAY FORMAT: Lots: X.XX (Risk $XXX.XX | SL: XX pips)
      ObjectSetString(0, "lblLotSize_v56", OBJPROP_TEXT, 
                     StringFormat("Lots: %.2f (Risk $%.2f | SL: %.1f pips)", 
                                 m_calculatedLots, riskAmount, sl_pips));
   }
   else
   {
      // Fallback to basic calculation
      CalculateLotSizeBasic();
      return;
   }
   
   ObjectSetInteger(0, "lblLotSize_v56", OBJPROP_COLOR, C'0,100,0'); // Dark green
   ChartRedraw(0);
}

//+------------------------------------------------------------------+
//| CRITICAL FIX: TRADE EXECUTION USING LINE PRICES WITH BINDING   |
//+------------------------------------------------------------------+
void CTradingAssistantFixed::ExecuteTradeFromOverlay(void)
{
    // ENHANCED CLICK PROTECTION: 3 second debounce for trade execution
    static uint last_trade_time = 0;
    uint current_time = GetTickCount();
    
    if(current_time - last_trade_time < 3000) // 3 second protection
    {
        return; // Silent block - no spam logs
    }
    
    if(!IsNewOverlayVisible())
    {
        return;
    }
    
    string symbol = m_currentSymbol;
    double lots = m_calculatedLots;
    if(lots <= 0) lots = 0.01;
    
    // CRITICAL FIX: Normalize lot size to prevent precision errors
    lots = NormalizeDouble(lots, 2);
    
    // Get active overlay
    CTradingDialog* active_overlay = GetActiveOverlay();
    if(active_overlay == NULL) return;
    
    // CRITICAL FIX: Get prices from ACTUAL LINES not panel positions
    double main_price = GetLinePriceFromObject(active_overlay.m_main_panel.m_line_name);
    double sl_price = GetLinePriceFromObject(active_overlay.m_sl_panel.m_line_name);
    double tp_price = GetLinePriceFromObject(active_overlay.m_tp_panel.m_line_name);
    
    // DEBUG: Print line prices to verify they're being read correctly
    Print("DEBUG: Line prices - Main: ", main_price, " SL: ", sl_price, " TP: ", tp_price);
    
    // Fallback to panel prices if line prices are not available
    if(main_price == 0.0)
    {
        main_price = active_overlay.m_main_panel.GetPriceLevel();
        Print("DEBUG: Using panel main price: ", main_price);
    }
    if(sl_price == 0.0)
    {
        sl_price = active_overlay.m_sl_panel.GetPriceLevel();
        Print("DEBUG: Using panel SL price: ", sl_price);
    }
    if(tp_price == 0.0)
    {
        tp_price = active_overlay.m_tp_panel.GetPriceLevel();
        Print("DEBUG: Using panel TP price: ", tp_price);
    }
    
    double current_ask = SymbolInfoDouble(symbol, SYMBOL_ASK);
    double current_bid = SymbolInfoDouble(symbol, SYMBOL_BID);
    double point = SymbolInfoDouble(symbol, SYMBOL_POINT);
    
    // Determine trade direction based on SL position relative to main panel
    bool is_buy_trade = (sl_price < main_price); // SL below main = BUY
    bool is_sell_trade = (sl_price > main_price); // SL above main = SELL
    
    if(!is_buy_trade && !is_sell_trade)
    {
        return; // Invalid setup
    }
    
    // ENHANCED ORDER TYPE DETECTION WITH 1-CLICK TRADING
    ENUM_ORDER_TYPE order_type;
    double execution_price;
    double reference_price = is_buy_trade ? current_ask : current_bid;
    
    // 1-CLICK TRADING: Smaller tolerance for instant market execution (20 points)
    double market_tolerance = 20 * point;
    double price_difference = MathAbs(main_price - reference_price);
    
    if(is_buy_trade)
    {
        if(InpOneClickTrading && price_difference <= market_tolerance)
        {
            order_type = ORDER_TYPE_BUY;
            execution_price = current_ask;
        }
        else if(main_price > current_ask + market_tolerance)
        {
            order_type = ORDER_TYPE_BUY_STOP;
            execution_price = main_price;
        }
        else if(main_price < current_ask - market_tolerance)
        {
            order_type = ORDER_TYPE_BUY_LIMIT;
            execution_price = main_price;
        }
        else
        {
            order_type = ORDER_TYPE_BUY;
            execution_price = current_ask;
        }
    }
    else // is_sell_trade
    {
        if(InpOneClickTrading && price_difference <= market_tolerance)
        {
            order_type = ORDER_TYPE_SELL;
            execution_price = current_bid;
        }
        else if(main_price < current_bid - market_tolerance)
        {
            order_type = ORDER_TYPE_SELL_STOP;
            execution_price = main_price;
        }
        else if(main_price > current_bid + market_tolerance)
        {
            order_type = ORDER_TYPE_SELL_LIMIT;
            execution_price = main_price;
        }
        else
        {
            order_type = ORDER_TYPE_SELL;
            execution_price = current_bid;
        }
    }
    
    // Validate symbol and market conditions
    if(!SymbolSelect(symbol, true))
    {
        return;
    }
    
    // Enhanced validation for TP/SL logic
    if(is_buy_trade && tp_price <= main_price)
    {
        return;
    }
    if(is_sell_trade && tp_price >= main_price)
    {
        return;
    }
    
    // VERIFY: Final prices that will be used in trade
    Print("FINAL TRADE PRICES: Entry=", execution_price, " SL=", sl_price, " TP=", tp_price);
    
    // CRITICAL FIX: Set proper trading parameters
    m_trade.SetExpertMagicNumber(InpMagicNumber);
    m_trade.SetDeviationInPoints(InpSlippage);
    
    // CRITICAL FIX: Set appropriate fill type for the broker
    ENUM_ORDER_TYPE_FILLING fillType = ORDER_FILLING_FOK;
    
    // Check what fill types are allowed by the symbol
    long fillingMode = SymbolInfoInteger(symbol, SYMBOL_FILLING_MODE);
    if((fillingMode & SYMBOL_FILLING_IOC) == SYMBOL_FILLING_IOC)
    {
        fillType = ORDER_FILLING_IOC;
    }
    else if((fillingMode & SYMBOL_FILLING_FOK) == SYMBOL_FILLING_FOK)
    {
        fillType = ORDER_FILLING_FOK;
    }
    else
    {
        fillType = ORDER_FILLING_FOK;
    }
    
    m_trade.SetTypeFilling(fillType);
    
    // Prepare trade request
    MqlTradeRequest request;
    MqlTradeResult result;
    
    ZeroMemory(request);
    request.action = TRADE_ACTION_PENDING;
    if(order_type == ORDER_TYPE_BUY || order_type == ORDER_TYPE_SELL)
    {
        request.action = TRADE_ACTION_DEAL; // Market order
    }
    
    request.symbol = symbol;
    request.volume = lots;
    request.type = order_type;
    request.price = execution_price;
    request.sl = sl_price;    // FIXED: Using actual line price
    request.tp = tp_price;    // FIXED: Using actual line price
    request.type_filling = fillType;
    
    // Enhanced deviation for market orders
    if(order_type == ORDER_TYPE_BUY || order_type == ORDER_TYPE_SELL)
    {
        request.deviation = 10; // Larger deviation for market orders
    }
    else
    {
        request.deviation = 3;  // Standard deviation for pending orders
    }
    
    request.magic = InpMagicNumber;
    request.comment = "Apeiron Edge v56.03";
    
    // Execute trade
    bool success = m_trade.OrderSend(request, result);
    last_trade_time = current_time; // Update debounce timer
    
    if(success && (result.retcode == TRADE_RETCODE_DONE || result.retcode == TRADE_RETCODE_PLACED))
    {
        // ONLY ORDER EXECUTION LOG: One line like your example
        string trade_type = is_buy_trade ? "Buy" : "Sell";
        Print("#", trade_type, " ", DoubleToString(lots, 2), " ", symbol, 
              " at: ", DoubleToString(execution_price, _Digits), 
              " tp: ", DoubleToString(tp_price, _Digits), 
              " sl: ", DoubleToString(sl_price, _Digits), " ok");
        
        // Sound alert for successful trade
        if(InpSoundAlerts)
        {
            PlaySound("ok.wav");
        }
        
        // CRITICAL: Bind trade to active overlay
        if(result.order > 0)
        {
            BindTradeToActiveOverlay(result.order);
        }
    }
    else
    {
        // ONLY TRADE FAILURE LOG
        Print("Trade failed: ", result.retcode, " - ", m_trade.ResultRetcodeDescription());
        
        // Sound alert for failed trade
        if(InpSoundAlerts)
        {
            PlaySound("timeout.wav");
        }
    }
}

//+------------------------------------------------------------------+
//| ENHANCED ORDER TYPE STRING FOR DISPLAY USING LINE PRICES       |
//+------------------------------------------------------------------+
string CTradingAssistantFixed::GetOrderTypeString(void)
{
    if(!IsNewOverlayVisible())
    {
        return "NO OVERLAY";
    }
    
    // Get active overlay
    CTradingDialog* active_overlay = GetActiveOverlay();
    if(active_overlay == NULL) return "NO OVERLAY";
    
    // CRITICAL FIX: Get prices from ACTUAL LINES not panel positions
    double main_price = GetLinePriceFromObject(active_overlay.m_main_panel.m_line_name);
    double sl_price = GetLinePriceFromObject(active_overlay.m_sl_panel.m_line_name);
    
    // Fallback to panel prices if line prices are not available
    if(main_price == 0.0)
    {
        main_price = active_overlay.m_main_panel.GetPriceLevel();
    }
    if(sl_price == 0.0)
    {
        sl_price = active_overlay.m_sl_panel.GetPriceLevel();
    }
    
    double current_ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
    double current_bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
    double point = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
    
    // Determine trade direction based on SL position relative to main panel
    bool is_buy_trade = (sl_price < main_price); // SL below main = BUY
    bool is_sell_trade = (sl_price > main_price); // SL above main = SELL
    
    if(!is_buy_trade && !is_sell_trade)
    {
        return "SL=ENTRY";
    }
    
    // Use proper reference price for each direction
    double reference_price = is_buy_trade ? current_ask : current_bid;
    double market_tolerance = 20 * point;
    double price_difference = MathAbs(main_price - reference_price);
    
    // Determine order type display logic
    if(is_buy_trade)
    {
        if(InpOneClickTrading && price_difference <= market_tolerance)
        {
            return "1-CLICK BUY";
        }
        else if(main_price > current_ask + market_tolerance)
        {
            return "BUY STOP";  // Entry above current price
        }
        else if(main_price < current_ask - market_tolerance)
        {
            return "BUY LIMIT"; // Entry below current price
        }
        else
        {
            return "BUY MARKET";
        }
    }
    else // is_sell_trade
    {
        if(InpOneClickTrading && price_difference <= market_tolerance)
        {
            return "1-CLICK SELL";
        }
        else if(main_price < current_bid - market_tolerance)
        {
            return "SELL STOP";  // Entry below current price
        }
        else if(main_price > current_bid + market_tolerance)
        {
            return "SELL LIMIT"; // Entry above current price
        }
        else
        {
            return "SELL MARKET";
        }
    }
}

//+------------------------------------------------------------------+
//| Calculate TP Panel Information Display                          |
//+------------------------------------------------------------------+
string CTradingAssistantFixed::CalculateTPInfo(double tpPrice, double entryPrice)
{
    if(!InpShowTPPanelInfo)
    {
        return DoubleToString(tpPrice, _Digits); // Only show price
    }
    
    double pips_distance = CalculatePipsDistance(tpPrice, entryPrice);
    double dollar_amount = CalculateDollarAmount(MathAbs(tpPrice - entryPrice));
    
    // TP Panel Format: "XX pts | $XX.XX | PRICE"
    return StringFormat("%.0f pts | $%.2f | %.5f", 
                       MathAbs(pips_distance * 10), // Convert to points
                       MathAbs(dollar_amount), // Profit amount
                       tpPrice);
}

//| Calculate SL Panel Information Display                          |
//+------------------------------------------------------------------+
string CTradingAssistantFixed::CalculateSLInfo(double slPrice, double entryPrice)

    if(!InpShowSLPanelInfo)
    {
        return DoubleToString(slPrice, _Digits); // Only show price
    }
    
    double pips_distance = CalculatePipsDistance(slPrice, entryPrice);
    double dollar_amount = CalculateDollarAmount(MathAbs(slPrice - entryPrice));
    
    // SL Panel Format: "XX pts | $XX.XX | PRICE"
    return StringFormat("%.0f pts | $%.2f | %.5f", 
                       MathAbs(pips_distance * 10), // Convert to points
                       MathAbs(dollar_amount), // Risk amount
                       slPrice);
}

//+------------------------------------------------------------------+
//| Calculate distance in pips between two prices                   |
//+------------------------------------------------------------------+
double CTradingAssistantFixed::CalculatePipsDistance(double price1, double price2)
{
    double price_diff = MathAbs(price1 - price2);
    double pip_size = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
    
    if(pip_size > 0)
    {
        return price_diff / pip_size / 10.0; // Convert to pips (10 points = 1 pip for most pairs)
    }
    return 0;
}

//+------------------------------------------------------------------+
//| Calculate dollar amount for price distance                      |
//+------------------------------------------------------------------+
double CTradingAssistantFixed::CalculateDollarAmount(double priceDistance)
{
    double pip_size = SymbolInfoDouble(_Symbol, SYMBOL_POINT);
    double tick_value = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
    double tick_size = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_SIZE);
    
    // Get current calculated lot size
    double lots = m_calculatedLots;
    if(lots <= 0) lots = 0.01;
    
    if(tick_value > 0 && tick_size > 0 && pip_size > 0)
    {
        double value_per_lot = tick_value * (priceDistance / tick_size);
        return value_per_lot * lots;
    }
    
    // Fallback calculation
    return priceDistance * lots * 100000; // Rough estimate
}

//| FIXED: Handle events - EVENT CONSUMPTION PATTERN               |
//+------------------------------------------------------------------+
bool CBaseTradingPanel::OnEvent(const int id, const long& lparam, 
                               const double& dparam, const string& sparam)
{
    switch(id)
    {
        case CHARTEVENT_MOUSE_MOVE:
        {
            int mouse_x = (int)lparam;
            int mouse_y = (int)dparam;
            int mouse_flags = (int)sparam;
            bool left_button_down = (mouse_flags & 1) > 0;
            
            // Check for button clicks - only on button DOWN, not during drag
            string clicked_button = "";
            if(left_button_down && !m_is_dragging && IsButtonClicked(mouse_x, mouse_y, clicked_button))
            {
                OnButtonClick(clicked_button);
                return true; // FIXED: Consume event to prevent duplicate processing
            }
            
            break;
        }
        
        case CHARTEVENT_OBJECT_CLICK:
        {
            // Handle direct button clicks for better responsiveness
            string clicked_object = sparam;
            for(int i = 0; i < 4; i++)
            {
                if(m_btn_names[i] == clicked_object)
                {
                    OnButtonClick(clicked_object);
                    return true; // FIXED: Consume event to prevent duplicate processing
                }
            }
            break;
        }
        
        case CHARTEVENT_CHART_CHANGE:
        {
            // Sync panel position with price level when chart changes
            if(!m_is_dragging)
            {
                SyncWithPriceLevel();
                UpdateButtonPositions();
            }
            break;
        }
    }
    
    // Forward to base class only if not handled
    return CWndClient::OnEvent(id, lparam, dparam, sparam);
}

void CBaseTradingPanel::UpdatePriceLevel(void)
{
    int chart_height = (int)ChartGetInteger(0, CHART_HEIGHT_IN_PIXELS);
    double chart_max = ChartGetDouble(0, CHART_PRICE_MAX);
    double chart_min = ChartGetDouble(0, CHART_PRICE_MIN);
    
    if(chart_height > 0 && chart_max > chart_min)

        int panel_y = Top();
        
        double price_range = chart_max - chart_min;
        double relative_position = (double)panel_y / chart_height;
        m_price_level = chart_max - (price_range * relative_position);
    }
}

        // Update button positions
        UpdateButtonPositions();
    }
}

void CMainTradingPanel::OnButtonClick(string button_name)
{
    if(button_name == m_btn_names[0])  // R button
    {
        // CRITICAL FIX: Disable R button in post-trade states
        if(m_parent_dialog != NULL)
        {
            ENUM_OVERLAY_STATE overlay_state = m_parent_dialog.GetOverlayState();
            if(overlay_state == OVERLAY_STATE_ACTIVE || overlay_state == OVERLAY_STATE_COLLAPSED)
            {
                // POST-TRADE: R button disabled to prevent moving TP/SL from actual trade levels
                Print("📋 R button disabled in post-trade mode - TP/SL are bound to actual trade");
                return;
            }
        }
        
        // PRE-TRADE: Normal R button functionality
        uint current_time = GetTickCount();
        if(current_time - m_last_flip_time > 500)
        {
            if(m_parent_dialog != NULL)
            {
                m_parent_dialog.FlipTPSLPanels();
                m_parent_dialog.UpdateAllLinesWithDynamicPositioning();
            }
            m_last_flip_time = current_time;
        }
    }
    else if(button_name == m_btn_names[1])  // M button
    {
        // ENHANCED: Different behavior based on trade state
        if(m_parent_dialog != NULL)
        {
            ENUM_OVERLAY_STATE overlay_state = m_parent_dialog.GetOverlayState();
            
            if(overlay_state == OVERLAY_STATE_ACTIVE || overlay_state == OVERLAY_STATE_COLLAPSED)
            {
                // POST-TRADE: M button toggles collapse/expand
                Print("📋 M button clicked - toggling collapse mode");
                m_parent_dialog.ToggleCollapseMode();
            }
            else
            {
                // PRE-TRADE: M button toggles price following
                TogglePriceFollowing();
            }
        }
        else
        {
            // Fallback to price following if no parent dialog
            TogglePriceFollowing();
        }
    }
    else if(button_name == m_btn_names[2])  // Price button - handle for dragging
    {
        // No action needed for dragging - handled in dialog movement system
    }
    else if(button_name == m_btn_names[3])  // Place/Close button
    {
        // FIXED: Add debounce to prevent multiple executions
        static uint last_action_time = 0;
        uint current_time = GetTickCount();
        if(current_time - last_action_time > 1000) // 1 second debounce
        {
            if(m_parent_dialog != NULL)
            {
                ENUM_OVERLAY_STATE overlay_state = m_parent_dialog.GetOverlayState();
                
                if(overlay_state == OVERLAY_STATE_ACTIVE || overlay_state == OVERLAY_STATE_COLLAPSED)
                {
                    // POST-TRADE: Close the position/order
                    Print("📋 Close button clicked");
                    CloseTradeFromOverlay();
                }
                else
                {
                    // PRE-TRADE: Execute new trade WITH PROPER BINDING
                    Print("📋 Place button clicked - executing trade with binding");
                    
                    // Get trade ticket from enhanced execution method
                    ulong trade_ticket = g_TradingAssistant.ExecuteTradeFromOverlayWithBinding();
                    
                    // CRITICAL FIX: Immediately bind if trade was successful
                    if(trade_ticket > 0)
                    {
                        Print("📋 Trade ", trade_ticket, " executed successfully - binding to overlay");
                        
                        // Bind trade to overlay
                        m_parent_dialog.BindToTrade(trade_ticket);
                        
                        // Force immediate update to post-trade layout
                        Sleep(100); // Small delay for trade to register
                        UpdateButtonPositions();
                        
                        Print("📋 Overlay bound to trade ", trade_ticket, " - state: ", EnumToString(m_parent_dialog.GetOverlayState()));
                    }
                    else
                    {
                        Print("⚠️ Trade execution failed or returned no ticket");
                    }
                }
            }
            else
            {
                // Fallback to normal trade execution
                Print("📋 Fallback trade execution - no parent dialog");
                g_TradingAssistant.ExecuteTradeFromOverlay();
            }
            
            last_action_time = current_time;
        }
    }
}

//+------------------------------------------------------------------+
//| Update position to follow current price                        |
//+------------------------------------------------------------------+
void CMainTradingPanel::UpdatePriceFollowing(void)
{
    if(!m_is_price_following) return;
    
    // Disable in post-trade states
    if(m_parent_dialog != NULL)
    {
        ENUM_OVERLAY_STATE overlay_state = m_parent_dialog.GetOverlayState();
        if(overlay_state == OVERLAY_STATE_ACTIVE || overlay_state == OVERLAY_STATE_COLLAPSED)
        {
            m_is_price_following = false;
            ObjectSetInteger(0, m_btn_names[1], OBJPROP_BGCOLOR, clrGreen);
            return;
        }
    }
    
    double current_price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
    
    // Calculate Y position for current price
    int chart_height = (int)ChartGetInteger(0, CHART_HEIGHT_IN_PIXELS);
    double chart_max = ChartGetDouble(0, CHART_PRICE_MAX);
    double chart_min = ChartGetDouble(0, CHART_PRICE_MIN);
    
    if(chart_height > 0 && chart_max > chart_min)
    {
        double price_range = chart_max - chart_min;
        double relative_position = (chart_max - current_price) / price_range;
        int new_y = (int)(relative_position * chart_height) - 1;
        
        // Calculate movement offset
        int movement_offset = new_y - Top();
        
        // Move main panel to current price
        int current_x = Left();
        Move(current_x, new_y);
        
        // Update price level and line with dynamic positioning
        m_price_level = current_price;
        UpdateDynamicHorizontalLine();
        UpdateButtonPositions();
        
        // Move TP/SL panels maintaining relative distance (only if not locked)
        if(m_parent_dialog != NULL)
        {
            // Get TP/SL panels and move them by the same offset
            CTPPanel* tp_panel = &(m_parent_dialog.m_tp_panel);
            CSLPanel* sl_panel = &(m_parent_dialog.m_sl_panel);
            
            if(!tp_panel.IsLocked())
            {
                tp_panel.Move(tp_panel.Left(), tp_panel.Top() + movement_offset);
                tp_panel.UpdatePriceLevel();
                tp_panel.UpdateDynamicHorizontalLine();
                tp_panel.UpdateButtonPositions();
            }
            
            if(!sl_panel.IsLocked())
            {
                sl_panel.Move(sl_panel.Left(), sl_panel.Top() + movement_offset);
                sl_panel.UpdatePriceLevel();
                sl_panel.UpdateDynamicHorizontalLine();
                sl_panel.UpdateButtonPositions();
            }
            
            // Update all lines with dynamic positioning
            m_parent_dialog.UpdateAllLinesWithDynamicPositioning();
            
            // Update shading areas
            m_parent_dialog.UpdateShadingAreas();
            
            // APEIRON INTEGRATION: Update lot size calculation when price following
            g_TradingAssistant.CalculateLotSize();
        }
    }
}

//+------------------------------------------------------------------+
//| Check if panel moved away from price following                 |
//+------------------------------------------------------------------+
void CMainTradingPanel::CheckPriceFollowingStatus(void)
{
    if(!m_is_price_following) return;
    
    double current_price = SymbolInfoDouble(_Symbol, SYMBOL_BID);
    
    int chart_height = (int)ChartGetInteger(0, CHART_HEIGHT_IN_PIXELS);
    double chart_max = ChartGetDouble(0, CHART_PRICE_MAX);
    double chart_min = ChartGetDouble(0, CHART_PRICE_MIN);
    
    if(chart_height > 0 && chart_max > chart_min)
    {
        double price_range = chart_max - chart_min;
        double relative_position = (chart_max - current_price) / price_range;
        int expected_y = (int)(relative_position * chart_height) - 1;
        
        int distance_from_price = MathAbs(Top() - expected_y);
        
        if(distance_from_price > 10)  // More than 10 pixels away = manually moved
        {
            m_is_price_following = false;
            ObjectSetInteger(0, m_btn_names[1], OBJPROP_BGCOLOR, clrGreen);  // Back to GREEN
            ChartRedraw(0);
        }
    }
}

//| FIXED: Close trade from overlay - PROPER CLEANUP               |
//+------------------------------------------------------------------+
void CMainTradingPanel::CloseTradeFromOverlay(void)
{
    if(m_parent_dialog == NULL) return;
    
    ulong ticket = m_parent_dialog.GetBoundTicket();
    if(ticket == 0) return;
    
    CTrade trade;
    trade.SetExpertMagicNumber(InpMagicNumber);
    
    bool trade_closed = false;
    
    // Check if it's a position or pending order
    if(PositionSelectByTicket(ticket))
    {
        if(trade.PositionClose(ticket))
        {
            trade_closed = true;
            if(InpSoundAlerts) PlaySound("ok.wav");
        }
        else
        {
            if(InpSoundAlerts) PlaySound("timeout.wav");
        }
    }
    else if(OrderSelect(ticket))
    {
        if(trade.OrderDelete(ticket))
        {
            trade_closed = true;
            if(InpSoundAlerts) PlaySound("ok.wav");
        }
        else
        {
            if(InpSoundAlerts) PlaySound("timeout.wav");
        }
    }
    
    // CLEAN: If trade closed successfully, destroy the overlay
    if(trade_closed)
    {
        if(m_parent_dialog != NULL)
        {
            for(int i = 0; i < g_overlay_count; i++)
            {
                if(&g_overlays[i] == m_parent_dialog)
                {
                    DestroyOverlay(i);
                    break;
                }
            }
        }
    }
}

    if(ticket == 0) return "NO TICKET";
    
    // Get trade information
    string trade_type = GetTradeTypeFromTicket(ticket);
    double lot_size = GetLotSizeFromTicket(ticket);
    double profit_loss = GetCurrentProfitFromTicket(ticket);
    
    ENUM_OVERLAY_STATE overlay_state = m_parent_dialog.GetOverlayState();
    
    if(overlay_state == OVERLAY_STATE_COLLAPSED)
    {
        return StringFormat("%s | %.2f | P&L: $%.2f", 
                           trade_type, lot_size, profit_loss);
    }
    else
    {
        return StringFormat("%s | %.2f | P&L: $%.2f | %.5f", 
                           trade_type, lot_size, profit_loss, m_price_level);
    }
}

//+------------------------------------------------------------------+
//| Get trade type string from ticket                              |
//+------------------------------------------------------------------+
string CMainTradingPanel::GetTradeTypeFromTicket(ulong ticket)
{
    if(PositionSelectByTicket(ticket))
    {
        ENUM_POSITION_TYPE pos_type = (ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE);
        return (pos_type == POSITION_TYPE_BUY) ? "BUY" : "SELL";
    }
    
    if(OrderSelect(ticket))
    {
        ENUM_ORDER_TYPE order_type = (ENUM_ORDER_TYPE)OrderGetInteger(ORDER_TYPE);
        switch(order_type)
        {
            case ORDER_TYPE_BUY_LIMIT:  return "BUY LIMIT";
            case ORDER_TYPE_SELL_LIMIT: return "SELL LIMIT";
            case ORDER_TYPE_BUY_STOP:   return "BUY STOP";
            case ORDER_TYPE_SELL_STOP:  return "SELL STOP";
            default: return "PENDING";
        }
    }
    
    return "UNKNOWN";
}

//+------------------------------------------------------------------+
//| Get lot size from ticket                                       |
//+------------------------------------------------------------------+
double CMainTradingPanel::GetLotSizeFromTicket(ulong ticket)
{
    if(PositionSelectByTicket(ticket))
        return PositionGetDouble(POSITION_VOLUME);
    
    if(OrderSelect(ticket))
        return OrderGetDouble(ORDER_VOLUME_CURRENT);
    
    return 0.0;
}

//+------------------------------------------------------------------+
//| Get current profit/loss from ticket                            |
//+------------------------------------------------------------------+
double CMainTradingPanel::GetCurrentProfitFromTicket(ulong ticket)
{
    if(PositionSelectByTicket(ticket))
        return PositionGetDouble(POSITION_PROFIT);
    
    return 0.0; // Pending orders have no P&L
}

// END OF PART 3A-6 v56.04 - DIALOG CONTAINER WITH PROPER PANEL ORDERING
//+------------------------------------------------------------------+
//|                     APEIRON EDGE v56 - PART 3A-7 FIXED         |
//|                DIALOG CONTAINER - EVENT HANDLING & MOVEMENT     |
//+------------------------------------------------------------------+

//+------------------------------------------------------------------+
//| FIXED: Handle dialog events - PROPER CONTAINER MOVEMENT        |
//+------------------------------------------------------------------+
bool CTradingDialog::OnEvent(const int id, const long& lparam, 
                            const double& dparam, const string& sparam)
{
    switch(id)
    {
        case CHARTEVENT_MOUSE_MOVE:
        {
            int mouse_x = (int)lparam;
            int mouse_y = (int)dparam;
            int mouse_flags = (int)sparam;
            bool left_button_down = (mouse_flags & 1) > 0;
            
            static bool chart_objects_disabled = false;
            
            // Handle main panel dragging (moves entire dialog as SOLID UNIT)
            if(left_button_down && m_main_panel.IsPriceButtonClicked(mouse_x, mouse_y) && !m_is_dragging)
            {
                // Check if no other panel is dragging
                if(!m_tp_panel.m_is_dragging && !m_sl_panel.m_is_dragging)
                {
                    // Check if main panel can be moved based on overlay state
                    bool can_move_main = false;
                    
                    if(m_overlay_state == OVERLAY_STATE_PRE_TRADE)
                    {
                        // Pre-trade: Main panel can move freely
                        can_move_main = true;
                    }
                    else if(m_overlay_state == OVERLAY_STATE_ACTIVE || m_overlay_state == OVERLAY_STATE_COLLAPSED)
                    {
                        // Bound overlay: Check if it's a pending order
                        if(m_bound_ticket > 0 && OrderSelect(m_bound_ticket))
                        {
                            // Pending orders: Main panel can move (modifies order price)
                            can_move_main = true;
                        }
                        else if(m_bound_ticket > 0 && PositionSelectByTicket(m_bound_ticket))
                        {
                            // Active position: Main panel CANNOT move (position already filled)
                            can_move_main = false;
                        }
                    }
                    
                    if(can_move_main)
                    {
                        m_is_dragging = true;
                        m_drag_offset_x = mouse_x - m_main_panel.Left();
                        m_drag_offset_y = mouse_y - m_main_panel.Top();
                        
                        // Disable chart scrolling during drag
                        ChartSetInteger(0, CHART_MOUSE_SCROLL, false);
                        ChartSetInteger(0, CHART_DRAG_TRADE_LEVELS, false);
                        chart_objects_disabled = true;
                        
                        // Set visual feedback
                        m_main_panel.SetDragHighlight(true);
                    }
                }
            }
            
            // Handle main panel drag release
            if(!left_button_down && m_is_dragging)
            {
                m_is_dragging = false;
                
                // Re-enable chart functionality
                ChartSetInteger(0, CHART_MOUSE_SCROLL, true);
                ChartSetInteger(0, CHART_DRAG_TRADE_LEVELS, true);
                chart_objects_disabled = false;
                
                m_main_panel.SetDragHighlight(false);
                
                // Check if price following should be disabled
                m_main_panel.CheckPriceFollowingStatus();
                
                // Update lot size after move
                g_TradingAssistant.CalculateLotSize();
                
                // FIXED: Force clean redraw to remove ghost buttons
                ChartRedraw(0);
                Sleep(10);
                ChartRedraw(0);
            }
            
            // FIXED: Handle main panel dragging movement with proper pending order handling
            if(m_is_dragging && left_button_down)
            {
                int new_x = mouse_x - m_drag_offset_x;
                int new_y = mouse_y - m_drag_offset_y;
                
                // Clamp to chart boundaries
                int chart_width = (int)ChartGetInteger(0, CHART_WIDTH_IN_PIXELS);
                int chart_height = (int)ChartGetInteger(0, CHART_HEIGHT_IN_PIXELS);
                
                if(new_x < 0) new_x = 0;
                if(new_y < 0) new_y = 0;
                if(new_x + 289 > chart_width) new_x = chart_width - 289;
                if(new_y + 25 > chart_height) new_y = chart_height - 25;
                
                // Calculate movement offsets
                int x_offset = new_x - m_main_panel.Left();
                int y_offset = new_y - m_main_panel.Top();
                
                // Always move main panel
                m_main_panel.Move(new_x, new_y);
                
                // FIXED: If bound to pending order, modify order entry price AND move TP/SL
                if((m_overlay_state == OVERLAY_STATE_ACTIVE || m_overlay_state == OVERLAY_STATE_COLLAPSED) && 
                   m_bound_ticket > 0 && OrderSelect(m_bound_ticket))
                {
                    // Update price level and modify pending order (this will move TP/SL automatically)
                    m_main_panel.UpdatePriceLevel();
                    ModifyTradeEntry(m_main_panel.GetPriceLevel());
                }
                else
                {
                    // FIXED: Handle TP/SL panels based on lock status for non-bound overlays
                    if(!m_tp_panel.IsLocked())
                    {
                        m_tp_panel.Move(m_tp_panel.Left() + x_offset, m_tp_panel.Top() + y_offset);
                    }
                    else
                    {
                        m_tp_panel.Move(m_tp_panel.Left() + x_offset, m_tp_panel.Top());
                    }
                    
                    if(!m_sl_panel.IsLocked())
                    {
                        m_sl_panel.Move(m_sl_panel.Left() + x_offset, m_sl_panel.Top() + y_offset);
                    }
                    else
                    {
                        m_sl_panel.Move(m_sl_panel.Left() + x_offset, m_sl_panel.Top());
                    }
                }
                
                OnMainPanelMove(new_x, new_y);
                g_TradingAssistant.CalculateLotSize();
            }
            
            // Handle individual TP panel dragging (vertical only)
            if(left_button_down && m_tp_panel.IsPriceButtonClicked(mouse_x, mouse_y) && 
               !m_tp_panel.m_is_dragging)
            {
                if(!m_is_dragging && !m_sl_panel.m_is_dragging)
                {
                    m_tp_panel.m_is_dragging = true;
                    m_tp_panel.m_drag_offset_y = mouse_y - m_tp_panel.Top();
                    
                    ChartSetInteger(0, CHART_MOUSE_SCROLL, false);
                    ChartSetInteger(0, CHART_DRAG_TRADE_LEVELS, false);
                    chart_objects_disabled = true;
                    
                    m_tp_panel.SetDragHighlight(true);
                }
            }
            
            if(!left_button_down && m_tp_panel.m_is_dragging)
            {
                m_tp_panel.m_is_dragging = false;
                
                ChartSetInteger(0, CHART_MOUSE_SCROLL, true);
                ChartSetInteger(0, CHART_DRAG_TRADE_LEVELS, true);
                chart_objects_disabled = false;
                
                m_tp_panel.SetDragHighlight(false);
                
                // FIXED: Force clean redraw to remove ghost buttons
                ChartRedraw(0);
                Sleep(10);
                ChartRedraw(0);
            }
            
            // TP panel movement - modify actual trade TP when bound
            if(m_tp_panel.m_is_dragging && left_button_down)
            {
                int new_y = mouse_y - m_tp_panel.m_drag_offset_y;
                
                int chart_height = (int)ChartGetInteger(0, CHART_HEIGHT_IN_PIXELS);
                if(new_y < 0) new_y = 0;
                if(new_y + 25 > chart_height) new_y = chart_height - 25;
                
                // Move TP panel
                m_tp_panel.Move(m_tp_panel.Left(), new_y);
                m_tp_panel.UpdatePriceLevel();
                m_tp_panel.UpdateDynamicHorizontalLine();
                m_tp_panel.UpdateButtonPositions();
                
                // If bound to trade, modify actual trade's TP
                if(m_overlay_state == OVERLAY_STATE_ACTIVE || m_overlay_state == OVERLAY_STATE_COLLAPSED)
                {
                    ModifyTradeTP(m_tp_panel.GetPriceLevel());
                }
                
                g_TradingAssistant.CalculateLotSize();
                
                // Update shading areas less frequently
                static int last_tp_y = -1;
                if(MathAbs(new_y - last_tp_y) > 3)
                {
                    OnTPPanelMove(new_y);
                    last_tp_y = new_y;
                }
            }
            
            // Handle individual SL panel dragging (vertical only)
            if(left_button_down && m_sl_panel.IsPriceButtonClicked(mouse_x, mouse_y) && 
               !m_sl_panel.m_is_dragging)
            {
                if(!m_is_dragging && !m_tp_panel.m_is_dragging)
                {
                    m_sl_panel.m_is_dragging = true;
                    m_sl_panel.m_drag_offset_y = mouse_y - m_sl_panel.Top();
                    
                    ChartSetInteger(0, CHART_MOUSE_SCROLL, false);
                    ChartSetInteger(0, CHART_DRAG_TRADE_LEVELS, false);
                    chart_objects_disabled = true;
                    
                    m_sl_panel.SetDragHighlight(true);
                }
            }
            
            if(!left_button_down && m_sl_panel.m_is_dragging)
            {
                m_sl_panel.m_is_dragging = false;
                
                ChartSetInteger(0, CHART_MOUSE_SCROLL, true);
                ChartSetInteger(0, CHART_DRAG_TRADE_LEVELS, true);
                chart_objects_disabled = false;
                
                m_sl_panel.SetDragHighlight(false);
                g_TradingAssistant.CalculateLotSize();
                
                // FIXED: Force clean redraw to remove ghost buttons
                ChartRedraw(0);
                Sleep(10);
                ChartRedraw(0);
            }
            
            // SL panel movement - modify actual trade SL when bound
            if(m_sl_panel.m_is_dragging && left_button_down)
            {
                int new_y = mouse_y - m_sl_panel.m_drag_offset_y;
                
                int chart_height = (int)ChartGetInteger(0, CHART_HEIGHT_IN_PIXELS);
                if(new_y < 0) new_y = 0;
                if(new_y + 25 > chart_height) new_y = chart_height - 25;
                
                // Move SL panel
                m_sl_panel.Move(m_sl_panel.Left(), new_y);
                m_sl_panel.UpdatePriceLevel();
                m_sl_panel.UpdateDynamicHorizontalLine();
                m_sl_panel.UpdateButtonPositions();
                
                // If bound to trade, modify actual trade's SL
                if(m_overlay_state == OVERLAY_STATE_ACTIVE || m_overlay_state == OVERLAY_STATE_COLLAPSED)
                {
                    ModifyTradeSL(m_sl_panel.GetPriceLevel());
                }
                
                g_TradingAssistant.CalculateLotSize();
                
                // Update shading areas less frequently
                static int last_sl_y = -1;
                if(MathAbs(new_y - last_sl_y) > 3)
                {
                    OnSLPanelMove(new_y);
                    last_sl_y = new_y;
                }
            }
            
            break;
        }
        
        case CHARTEVENT_OBJECT_CLICK:
        {
            // FIXED: Handle main panel click for collapse/expand in post-trade mode
            if(m_overlay_state == OVERLAY_STATE_ACTIVE && sparam == m_main_panel.m_btn_names[2])
            {
                // Clicked on main handle in active mode - toggle collapse
                ToggleCollapseMode();
                return true;
            }
            else if(m_overlay_state == OVERLAY_STATE_COLLAPSED && sparam == m_main_panel.m_btn_names[2])
            {
                // Clicked on main handle in collapsed mode - expand
                ToggleCollapseMode();
                return true;
            }
            break;
        }
        
        case CHARTEVENT_CHART_CHANGE:
        {
            // Simple chart change handling
            if(!m_is_dragging && !m_tp_panel.m_is_dragging && !m_sl_panel.m_is_dragging)
            {
                // Always sync main panel
                m_main_panel.SyncWithPriceLevel();
                
                // Only sync unlocked panels
                if(!m_tp_panel.IsLocked())
                {
                    m_tp_panel.SyncWithPriceLevel();
                }
                if(!m_sl_panel.IsLocked())
                {
                    m_sl_panel.SyncWithPriceLevel();
                }
                
                UpdateShadingAreas();
            }
            break;
        }
    }
    
    // Forward events to child panels
    bool main_handled = m_main_panel.OnEvent(id, lparam, dparam, sparam);
    bool tp_handled = m_tp_panel.OnEvent(id, lparam, dparam, sparam);
    bool sl_handled = m_sl_panel.OnEvent(id, lparam, dparam, sparam);
    
    if(main_handled || tp_handled || sl_handled)
    {
        return true;
    }
    
    // Forward to base class
    return CAppDialog::OnEvent(id, lparam, dparam, sparam);
}

//+------------------------------------------------------------------+
//| FIXED: Update system on each tick - CLEAN OVERLAY MANAGEMENT   |
//+------------------------------------------------------------------+
void CTradingDialog::OnTick(void)
{
    static int tick_counter = 0;
    static string last_symbol = "";
    static long last_chart_id = 0;
    
    string current_symbol = _Symbol;
    long current_chart_id = ChartID();
    
    tick_counter++;
    
    // FIXED: Check for scheduled destruction first
    if(m_needs_destruction)
    {
        // Find this overlay and destroy it
        for(int i = 0; i < g_overlay_count; i++)
        {
            if(&g_overlays[i] == GetPointer(this))
            {
                DestroyOverlay(i);
                return;
            }
        }
    }
    
    // Handle symbol/chart changes
    if((current_symbol != last_symbol || current_chart_id != last_chart_id) && 
       (last_symbol != "" || last_chart_id != 0))
    {
        Sleep(50);
        UpdateShadingAreas();
        UpdateAllLines();
        g_TradingAssistant.CalculateLotSize();
        ChartRedraw(0);
        last_symbol = current_symbol;
        last_chart_id = current_chart_id;
        return;
    }
    last_symbol = current_symbol;
    last_chart_id = current_chart_id;
    
    // Update trade information every 10 ticks for real-time P&L
    if(tick_counter % 10 == 0)
    {
        UpdateTradeInfo();
    }
    
    // Update price following every 10 ticks
    if(m_main_panel.IsPriceFollowing() && tick_counter % 10 == 0)
    {
        m_main_panel.UpdatePriceFollowing();
        UpdateShadingAreas();
    }
    
    // REMOVED: Excessive UpdateButtonPositions() calls every 100 ticks
    // Only update when layout actually changes
}

{
    if(!m_initialized)
        return;

    {
        ObjectDelete(0, m_tp_shading.m_area_name);
        m_tp_shading.m_area_name = "";
    }
    
    if(m_sl_shading.m_area_name != "")
    {
        ObjectDelete(0, m_sl_shading.m_area_name);
        m_sl_shading.m_area_name = "";
    }

    for(int i = 0; i < 4; i++)
    {
        if(m_main_panel.m_btn_names[i] != "")
        {
            ObjectDelete(0, m_main_panel.m_btn_names[i]);
            m_main_panel.m_btn_names[i] = "";
        }
        if(m_tp_panel.m_btn_names[i] != "")
        {
            ObjectDelete(0, m_tp_panel.m_btn_names[i]);
            m_tp_panel.m_btn_names[i] = "";
        }
        if(m_sl_panel.m_btn_names[i] != "")
        {
            ObjectDelete(0, m_sl_panel.m_btn_names[i]);
            m_sl_panel.m_btn_names[i] = "";
        }
    }
    
    // Delete horizontal lines
    if(m_main_panel.m_line_name != "")
    {
        ObjectDelete(0, m_main_panel.m_line_name);
        m_main_panel.m_line_name = "";
    }
    if(m_tp_panel.m_line_name != "")
    {
        ObjectDelete(0, m_tp_panel.m_line_name);
        m_tp_panel.m_line_name = "";
    }
    if(m_sl_panel.m_line_name != "")
    {
        ObjectDelete(0, m_sl_panel.m_line_name);
        m_sl_panel.m_line_name = "";
    }
    
    m_initialized = false;
    
    // Call base class destroy only if reason is REMOVE
    if(reason == REASON_REMOVE)
    {
        CAppDialog::Destroy(reason);
    }
    
    // FIXED: Force multiple redraws to clear ghost buttons
    ChartRedraw(0);
    Sleep(10);
    ChartRedraw(0);
    Sleep(10);
    ChartRedraw(0);
}

bool CTradingDialog::CreateChildPanels(void)
{
    int panel_x = InpOverlayPanelX;
    int panel_y = InpOverlayPanelY;
    
    // Create shading areas FIRST (background)
    int chart_width = (int)ChartGetInteger(0, CHART_WIDTH_IN_PIXELS);
    int chart_height = (int)ChartGetInteger(0, CHART_HEIGHT_IN_PIXELS);
    
    if(!m_tp_shading.Create(0, "TPShading", 0, 0, 0, chart_width, chart_height))
        return false;
    m_tp_shading.SetAreaType("TP");
    m_tp_shading.SetAreaColor(InpProfitFillColor);
    m_tp_shading.SetParentDialog(GetPointer(this));
    
    if(!m_sl_shading.Create(0, "SLShading", 0, 0, 0, chart_width, chart_height))
        return false;
    m_sl_shading.SetAreaType("SL");
    m_sl_shading.SetAreaColor(InpRiskFillColor);
    m_sl_shading.SetParentDialog(GetPointer(this));
    
    // Create TP panel SECOND
    if(!m_tp_panel.Create(0, "TPPanel", 0, panel_x, panel_y - 33, panel_x + 289, panel_y - 8))
        return false;
    m_tp_panel.SetParentDialog(GetPointer(this));
    
    // Create SL panel THIRD
    if(!m_sl_panel.Create(0, "SLPanel", 0, panel_x, panel_y + 33, panel_x + 289, panel_y + 58))
        return false;
    m_sl_panel.SetParentDialog(GetPointer(this));
    
    // Create main panel LAST (foreground)
    if(!m_main_panel.Create(0, "MainTradingPanel", 0, panel_x, panel_y, panel_x + 289, panel_y + 25))
        return false;
    m_main_panel.SetParentDialog(GetPointer(this));
    
    // Add all elements to dialog in proper order
    if(!Add(m_tp_shading)) return false;
    if(!Add(m_sl_shading)) return false;
    if(!Add(m_tp_panel)) return false;
    if(!Add(m_sl_panel)) return false;
    if(!Add(m_main_panel)) return false;
    
    return true;
}

//| Flip TP and SL panel positions                                 |
//+------------------------------------------------------------------+
void CTradingDialog::FlipTPSLPanels(void)
{
    int tp_distance = m_tp_panel.Top() - m_main_panel.Top();
    int sl_distance = m_sl_panel.Top() - m_main_panel.Top();
    
    int new_tp_distance = -tp_distance;
    int new_sl_distance = -sl_distance;
    
    m_tp_panel.Move(m_tp_panel.Left(), m_main_panel.Top() + new_tp_distance);
    m_tp_panel.UpdatePriceLevel();
    m_tp_panel.UpdateDynamicHorizontalLine();
    m_tp_panel.UpdateButtonPositions();
    
    m_sl_panel.Move(m_sl_panel.Left(), m_main_panel.Top() + new_sl_distance);
    m_sl_panel.UpdatePriceLevel();
    m_sl_panel.UpdateDynamicHorizontalLine();
    m_sl_panel.UpdateButtonPositions();

//+------------------------------------------------------------------+
//| FIXED: Modify pending order's entry price with TP/SL maintenance|
//+------------------------------------------------------------------+
void CTradingDialog::ModifyTradeEntry(double new_entry_price)
{
    if(m_bound_ticket == 0 || new_entry_price <= 0.0) return;
    
    // Only works for pending orders
    if(OrderSelect(m_bound_ticket))
    {
        CTrade trade;
        trade.SetExpertMagicNumber(InpMagicNumber);
        
        // FIXED: Maintain TP/SL distances when moving entry
        double old_entry = OrderGetDouble(ORDER_PRICE_OPEN);
        double old_sl = OrderGetDouble(ORDER_SL);
        double old_tp = OrderGetDouble(ORDER_TP);
        
        double new_sl = 0, new_tp = 0;
        
        if(old_sl > 0)
        {
            // Maintain SL distance
            double sl_distance = old_sl - old_entry;
            new_sl = new_entry_price + sl_distance;
        }
        
        if(old_tp > 0)
        {
            // Maintain TP distance  
            double tp_distance = old_tp - old_entry;
            new_tp = new_entry_price + tp_distance;
        }
        
        if(trade.OrderModify(m_bound_ticket, new_entry_price, new_sl, new_tp, ORDER_TIME_GTC, 0))
        {
            Print("✅ Pending order entry modified to: ", new_entry_price, " SL: ", new_sl, " TP: ", new_tp);
            
            // FIXED: Move TP/SL panels to maintain visual consistency
            if(new_sl > 0 && old_sl > 0)
            {
                double sl_price_diff = new_sl - old_sl;
                int sl_y_offset = ConvertPriceToPixelOffset(sl_price_diff);
                m_sl_panel.Move(m_sl_panel.Left(), m_sl_panel.Top() + sl_y_offset);
                m_sl_panel.UpdatePriceLevel();
                m_sl_panel.UpdateDynamicHorizontalLine();
            }
            
            if(new_tp > 0 && old_tp > 0)
            {
                double tp_price_diff = new_tp - old_tp;
                int tp_y_offset = ConvertPriceToPixelOffset(tp_price_diff);
                m_tp_panel.Move(m_tp_panel.Left(), m_tp_panel.Top() + tp_y_offset);
                m_tp_panel.UpdatePriceLevel();
                m_tp_panel.UpdateDynamicHorizontalLine();
            }
            
            UpdateShadingAreas();
        }
    }
}

//+------------------------------------------------------------------+
//| NEW: Convert price difference to pixel offset                  |
//+------------------------------------------------------------------+
int CTradingDialog::ConvertPriceToPixelOffset(double price_diff)
{
    int chart_height = (int)ChartGetInteger(0, CHART_HEIGHT_IN_PIXELS);
    double chart_max = ChartGetDouble(0, CHART_PRICE_MAX);
    double chart_min = ChartGetDouble(0, CHART_PRICE_MIN);
    
    if(chart_height > 0 && chart_max > chart_min)
    {
        double price_range = chart_max - chart_min;
        return (int)((price_diff / price_range) * chart_height * -1); // Negative because Y increases downward
    }
    
    return 0;
}

//+------------------------------------------------------------------+
//| Enhanced Trade Execution with Ticket Return                     |
//+------------------------------------------------------------------+
ulong CTradingAssistantFixed::ExecuteTradeFromOverlayWithBinding(void)
{
    // Enhanced click protection: 3 second debounce for trade execution
    static uint last_trade_time = 0;
    uint current_time = GetTickCount();
    
    if(current_time - last_trade_time < 3000) // 3 second protection
    {
        return 0; // Silent block - no spam logs
    }
    
    if(!IsNewOverlayVisible())
    {
        return 0;
    }
    
    string symbol = m_currentSymbol;
    double lots = m_calculatedLots;
    if(lots <= 0) lots = 0.01;
    
    // Normalize lot size to prevent precision errors
    lots = NormalizeDouble(lots, 2);
    
    // Get active overlay
    CTradingDialog* active_overlay = GetActiveOverlay();
    if(active_overlay == NULL) return 0;
    
    // Get prices from ACTUAL LINES not panel positions
    double main_price = GetLinePriceFromObject(active_overlay.m_main_panel.m_line_name);
    double sl_price = GetLinePriceFromObject(active_overlay.m_sl_panel.m_line_name);
    double tp_price = GetLinePriceFromObject(active_overlay.m_tp_panel.m_line_name);
    
    // Fallback to panel prices if line prices are not available
    if(main_price == 0.0)
        main_price = active_overlay.m_main_panel.GetPriceLevel();
    if(sl_price == 0.0)
        sl_price = active_overlay.m_sl_panel.GetPriceLevel();
    if(tp_price == 0.0)
        tp_price = active_overlay.m_tp_panel.GetPriceLevel();
    
    double current_ask = SymbolInfoDouble(symbol, SYMBOL_ASK);
    double current_bid = SymbolInfoDouble(symbol, SYMBOL_BID);
    double point = SymbolInfoDouble(symbol, SYMBOL_POINT);
    
    // Determine trade direction based on SL position relative to main panel
    bool is_buy_trade = (sl_price < main_price);
    bool is_sell_trade = (sl_price > main_price);
    
    if(!is_buy_trade && !is_sell_trade)
    {
        return 0; // Invalid setup
    }
    
    // Enhanced order type detection with 1-click trading
    ENUM_ORDER_TYPE order_type;
    double execution_price;
    double reference_price = is_buy_trade ? current_ask : current_bid;
    
    // 1-click trading: smaller tolerance for instant market execution
    double market_tolerance = 20 * point;
    double price_difference = MathAbs(main_price - reference_price);
    
    if(is_buy_trade)
    {
        if(InpOneClickTrading && price_difference <= market_tolerance)
        {
            order_type = ORDER_TYPE_BUY;
            execution_price = current_ask;
        }
        else if(main_price > current_ask + market_tolerance)
        {
            order_type = ORDER_TYPE_BUY_STOP;
            execution_price = main_price;
        }
        else if(main_price < current_ask - market_tolerance)
        {
            order_type = ORDER_TYPE_BUY_LIMIT;
            execution_price = main_price;
        }
        else
        {
            order_type = ORDER_TYPE_BUY;
            execution_price = current_ask;
        }
    }
    else // is_sell_trade
    {
        if(InpOneClickTrading && price_difference <= market_tolerance)
        {
            order_type = ORDER_TYPE_SELL;
            execution_price = current_bid;
        }
        else if(main_price < current_bid - market_tolerance)
        {
            order_type = ORDER_TYPE_SELL_STOP;
            execution_price = main_price;
        }
        else if(main_price > current_bid + market_tolerance)
        {
            order_type = ORDER_TYPE_SELL_LIMIT;
            execution_price = main_price;
        }
        else
        {
            order_type = ORDER_TYPE_SELL;
            execution_price = current_bid;
        }
    }
    
    // Validate symbol and market conditions
    if(!SymbolSelect(symbol, true))
    {
        return 0;
    }
    
    // Enhanced validation for TP/SL logic
    if(is_buy_trade && tp_price <= main_price)
    {
        return 0;
    }
    if(is_sell_trade && tp_price >= main_price)
    {
        return 0;
    }
    
    // Set proper trading parameters
    m_trade.SetExpertMagicNumber(InpMagicNumber);
    m_trade.SetDeviationInPoints(InpSlippage);
    
    // Set appropriate fill type for the broker
    ENUM_ORDER_TYPE_FILLING fillType = ORDER_FILLING_FOK;
    long fillingMode = SymbolInfoInteger(symbol, SYMBOL_FILLING_MODE);
    if((fillingMode & SYMBOL_FILLING_IOC) == SYMBOL_FILLING_IOC)
    {
        fillType = ORDER_FILLING_IOC;
    }
    else if((fillingMode & SYMBOL_FILLING_FOK) == SYMBOL_FILLING_FOK)
    {
        fillType = ORDER_FILLING_FOK;
    }
    
    m_trade.SetTypeFilling(fillType);
    
    // Prepare trade request
    MqlTradeRequest request;
    MqlTradeResult result;
    
    ZeroMemory(request);
    request.action = TRADE_ACTION_PENDING;
    if(order_type == ORDER_TYPE_BUY || order_type == ORDER_TYPE_SELL)
    {
        request.action = TRADE_ACTION_DEAL; // Market order
    }
    
    request.symbol = symbol;
    request.volume = lots;
    request.type = order_type;
    request.price = execution_price;
    request.sl = sl_price;
    request.tp = tp_price;
    request.type_filling = fillType;
    
    // Enhanced deviation for market orders
    if(order_type == ORDER_TYPE_BUY || order_type == ORDER_TYPE_SELL)
    {
        request.deviation = 10; // Larger deviation for market orders
    }
    else
    {
        request.deviation = 3;  // Standard deviation for pending orders
    }
    
    request.magic = InpMagicNumber;
    request.comment = "Apeiron Edge v56.03 - Overlay";
    
    // Execute trade
    bool success = m_trade.OrderSend(request, result);
    last_trade_time = current_time; // Update debounce timer
    
    if(success && (result.retcode == TRADE_RETCODE_DONE || result.retcode == TRADE_RETCODE_PLACED))
    {
        // ORDER EXECUTION LOG: One line format
        string trade_type = is_buy_trade ? "Buy" : "Sell";
        Print("#", trade_type, " ", DoubleToString(lots, 2), " ", symbol, 
              " at: ", DoubleToString(execution_price, _Digits), 
              " tp: ", DoubleToString(tp_price, _Digits), 
              " sl: ", DoubleToString(sl_price, _Digits), " ok");
        
        // Sound alert for successful trade
        if(InpSoundAlerts)
        {
            PlaySound("ok.wav");
        }
        
        // RETURN THE TICKET for binding
        if(result.order > 0)
        {
            return result.order;
        }
        else if(result.deal > 0)
        {
            // For market orders, we need to find the position ticket
            // This is usually the same as the order ticket, but let's be safe
            return result.order;
        }
    }
    else
    {
        // TRADE FAILURE LOG
        Print("Trade failed: ", result.retcode, " - ", m_trade.ResultRetcodeDescription());
        
        // Sound alert for failed trade
        if(InpSoundAlerts)
        {
            PlaySound("timeout.wav");
        }
    }
    
    return 0; // No ticket if failed
}

{

        return;
    
    Print("🗑️ Destroying overlay #", index);
    
    // Destroy the overlay
    g_overlays[index].Destroy(REASON_REMOVE);
    
    // FIXED: Shift remaining overlays down to fill the gap
    for(int i = index; i < g_overlay_count - 1; i++)
    {
        g_overlays[i] = g_overlays[i + 1];
    }
    
    // Clear the last slot
    g_overlays[g_overlay_count - 1] = CTradingDialog();
    
    // Decrease count
    g_overlay_count--;
    
    // FIXED: Update active overlay index
    if(g_active_overlay_index == index)
    {
        // Active overlay was destroyed, set to last overlay or -1 if none
        g_active_overlay_index = (g_overlay_count > 0) ? g_overlay_count - 1 : -1;
    }
    else if(g_active_overlay_index > index)
    {
        // Active overlay shifted down
        g_active_overlay_index--;
    }
    
    Print("✅ Overlay destroyed. Remaining: ", g_overlay_count);
    
    // Update overlay button state
    UpdateOverlayButtonState();
}

{
    for(int i = 0; i < g_overlay_count; i++)

        if(g_overlays[i].GetBoundTicket() == ticket)
        {
            return i;
        }
    }
    return -1;
}

//+------------------------------------------------------------------+
//| FIXED: Check and cleanup closed trades                         |
//+------------------------------------------------------------------+
void CheckAndCleanupClosedTrades(void)
{
    // Check from the end to avoid index issues when destroying
    for(int i = g_overlay_count - 1; i >= 0; i--)
    {
        if(g_overlays[i].m_initialized)
        {
            ulong ticket = g_overlays[i].GetBoundTicket();
            
            if(ticket > 0)
            {
                // Check if trade still exists
                bool trade_exists = false;
                
                if(PositionSelectByTicket(ticket))
                    trade_exists = true;
                else if(OrderSelect(ticket))
                    trade_exists = true;
                
                if(!trade_exists)
                {
                    Print("🧹 Trade ", ticket, " no longer exists, cleaning up overlay #", i);
                    DestroyOverlay(i);
                }
            }
        }
    }
}

//+------------------------------------------------------------------+
//| TRADING FUNCTIONS - CLEANED                                     |
//+------------------------------------------------------------------+
void CTradingAssistantFixed::OnTradeClick(bool isBuy)
{
   Print("=== EXECUTING TRADE v56: ", isBuy ? "BUY" : "SELL", " ===");
   
   string symbol = m_currentSymbol;
   double lots = m_calculatedLots;
   if(lots <= 0) lots = 0.01;
   
   // Validate symbol and market conditions
   if(!SymbolSelect(symbol, true))
   {
      Print("ERROR: Symbol ", symbol, " not available for trading");
      return;
   }
   
   // Get current prices
   double currentPrice = isBuy ? SymbolInfoDouble(symbol, SYMBOL_ASK) : SymbolInfoDouble(symbol, SYMBOL_BID);
   double point = SymbolInfoDouble(symbol, SYMBOL_POINT);
   double sl = 0, tp = 0;
   
   // Calculate SL and TP (50 points SL, 100 points TP)
   if(isBuy)
   {
      sl = currentPrice - (50 * point);
      tp = currentPrice + (100 * point);
   }
   else
   {
      sl = currentPrice + (50 * point);
      tp = currentPrice - (100 * point);
   }
   
   // Prepare trade request
   MqlTradeRequest request;
   MqlTradeResult result;
   
   ZeroMemory(request);
   request.action = TRADE_ACTION_DEAL;
   request.symbol = symbol;
   request.volume = lots;
   request.type = isBuy ? ORDER_TYPE_BUY : ORDER_TYPE_SELL;
   request.price = currentPrice;
   request.sl = sl;
   request.tp = tp;
   request.deviation = 3;
   request.magic = InpMagicNumber;
   request.comment = "Apeiron Edge v56";
   
   // Execute trade
   bool success = m_trade.OrderSend(request, result);
   if(success && result.retcode == TRADE_RETCODE_DONE)
   {
      Print("✅ TRADE SUCCESS v56: ", isBuy ? "BUY" : "SELL", " ", lots, " lots of ", symbol);
      Print("   Entry: ", currentPrice, " | SL: ", sl, " | TP: ", tp);
      Print("   Ticket: ", result.order);
      
      // Show overlay if not visible (for new overlay system)
      if(!IsNewOverlayVisible())
      {
         if(InitializeNewOverlaySystem())
         {
            ObjectSetInteger(0, "btnLines_v56", OBJPROP_BGCOLOR, C'0,128,0'); // Green
            ObjectSetString(0, "btnLines_v56_Text", OBJPROP_TEXT, "Lines ON");
            ChartRedraw(0);
         }
      }
   }
   else
   {
      Print("❌ TRADE FAILED v56: ", result.retcode, " - ", m_trade.ResultRetcodeDescription());
   }
}

void CTradingAssistantFixed::ClosePositions(int positionType)
{
   Print("=== CLOSING POSITIONS v56: Type ", positionType, " ===");
   
   string symbol = m_currentSymbol;
   int closedCount = 0;
   double totalVolume = 0;
   double totalProfit = 0;
   
   for(int i = PositionsTotal() - 1; i >= 0; i--)
   {
      if(PositionGetSymbol(i) == symbol)
      {
         long type = PositionGetInteger(POSITION_TYPE);
         
         // Check if we should close this position
         if(positionType == -1 || type == positionType)
         {
            ulong ticket = PositionGetInteger(POSITION_TICKET);
            double volume = PositionGetDouble(POSITION_VOLUME);
            double profit = PositionGetDouble(POSITION_PROFIT);
            
            MqlTradeRequest request;
            MqlTradeResult result;
            
            ZeroMemory(request);
            request.action = TRADE_ACTION_DEAL;
            request.symbol = symbol;
            request.volume = volume;
            request.type = (type == POSITION_TYPE_BUY) ? ORDER_TYPE_SELL : ORDER_TYPE_BUY;
            request.position = ticket;
            request.price = (type == POSITION_TYPE_BUY) ? SymbolInfoDouble(symbol, SYMBOL_BID) : SymbolInfoDouble(symbol, SYMBOL_ASK);
            request.deviation = 3;
            request.magic = InpMagicNumber;
            
            if(m_trade.OrderSend(request, result) && result.retcode == TRADE_RETCODE_DONE)
            {
               closedCount++;
               totalVolume += volume;
               totalProfit += profit;
               Print("✅ POSITION CLOSED: Ticket ", ticket, " Volume ", volume, " Profit $", profit);
            }
            else
            {
               Print("❌ CLOSE FAILED: Ticket ", ticket, " Error ", result.retcode);
            }
         }
      }
   }
   
   Print("=== CLOSE COMPLETE v56: ", closedCount, " positions closed ===");
   Print("   Total Volume: ", totalVolume, " | Total Profit: $", totalProfit);
}

bool CTradingAssistantFixed::OnEvent(const int id, const long &lparam, const double &dparam, const string &sparam)

    if(HandleDropdownEvents(id, lparam, dparam, sparam)) return true;
    if(HandleBrowserEvents(id, lparam, dparam, sparam)) return true;

        if(StringFind(sparam, "ShadingArea") >= 0)

            return false; // Silently ignore - no logging, no processing
        }

        if(StringFind(sparam, "Place") >= 0 || 
           StringFind(sparam, "MainBtn") >= 0 ||
           StringFind(sparam, "TPBtn") >= 0 ||
           StringFind(sparam, "SLBtn") >= 0 ||

        }

            return true;

void CTradingAssistantFixed::RemoveSymbolFromWatchlist(string symbol)
{
   for(int i = 0; i < m_watchlistCount; i++)
   {
      if(m_watchlistSymbols[i] == symbol)
      {
         // Shift remaining symbols down
         for(int j = i; j < m_watchlistCount - 1; j++)
         {
            m_watchlistSymbols[j] = m_watchlistSymbols[j + 1];
         }
         m_watchlistCount--;
         Print("Removed symbol from watchlist v56: ", symbol, " (Total: ", m_watchlistCount, ")");
         return;
      }
   }
   Print("Symbol not found in watchlist: ", symbol);
}

bool CTradingAssistantFixed::IsSymbolInWatchlist(string symbol)
{
   for(int i = 0; i < m_watchlistCount; i++)
   {
      if(m_watchlistSymbols[i] == symbol)
         return true;
   }
   return false;
}

void CTradingAssistantFixed::LoadBrokerSymbols(void)
{
   Print("=== LOADING MARKET WATCH SYMBOLS v56 ===");
   
   m_totalSymbols = 0;
   int marketWatchCount = SymbolsTotal(true); // true = only Market Watch symbols
   
   Print("Market Watch symbols available: ", marketWatchCount);
   
   // Load ONLY Market Watch symbols (fast, no MT slowdown)
   for(int i = 0; i < marketWatchCount && m_totalSymbols < 1000; i++)
   {
      string symbolName = SymbolName(i, true); // true = Market Watch only
      if(StringLen(symbolName) > 0)
      {
         m_allSymbols[m_totalSymbols] = symbolName;
         m_totalSymbols++;
      }
   }
   
   Print("=== LOADED ", m_totalSymbols, " MARKET WATCH SYMBOLS v56 ===");
   
   // Sort symbols alphabetically for better browsing
   for(int i = 0; i < m_totalSymbols - 1; i++)
   {
      for(int j = i + 1; j < m_totalSymbols; j++)
      {
         if(StringCompare(m_allSymbols[i], m_allSymbols[j]) > 0)
         {
            string temp = m_allSymbols[i];
            m_allSymbols[i] = m_allSymbols[j];
            m_allSymbols[j] = temp;
         }
      }
   }
   
   Print("✅ Market Watch symbols sorted alphabetically");
}

//+------------------------------------------------------------------+
//| ENHANCED FILTER WITH SEARCH ALL OPTION                        |
//+------------------------------------------------------------------+
void CTradingAssistantFixed::FilterBrowserSymbols(string searchText)
{
   Print("Filtering browser symbols v56: '", searchText, "'");
   
   StringToUpper(searchText);
   StringTrimLeft(searchText);
   StringTrimRight(searchText);
   
   m_filteredCount = 0;
   
   // If no search text, show Market Watch symbols
   if(StringLen(searchText) == 0)
   {
      for(int i = 0; i < m_totalSymbols && m_filteredCount < 50; i++)
      {
         m_filteredSymbols[m_filteredCount] = m_allSymbols[i];
         m_filteredCount++;
      }
   }
   else
   {
      // SEARCH: First check Market Watch symbols
      for(int i = 0; i < m_totalSymbols && m_filteredCount < 25; i++)
      {
         string symbol = m_allSymbols[i];
         string upperSymbol = symbol;
         StringToUpper(upperSymbol);
         
         if(StringFind(upperSymbol, searchText) >= 0)
         {
            m_filteredSymbols[m_filteredCount] = symbol;
            m_filteredCount++;
         }
      }
      
      // SEARCH: Then check ALL broker symbols if not enough results
      if(m_filteredCount < 10)
      {
         int totalAll = SymbolsTotal(false); // All broker symbols
         Print("Searching all ", totalAll, " broker symbols for: ", searchText);
         
         for(int i = 0; i < totalAll && m_filteredCount < 50; i++)
         {
            string symbolName = SymbolName(i, false); // All symbols
            if(StringLen(symbolName) > 0)
            {
               string upperSymbol = symbolName;
               StringToUpper(upperSymbol);
               
               if(StringFind(upperSymbol, searchText) >= 0)
               {
                  // Check if already added from Market Watch
                  bool alreadyAdded = false;
                  for(int j = 0; j < m_filteredCount; j++)
                  {
                     if(m_filteredSymbols[j] == symbolName)
                     {
                        alreadyAdded = true;
                        break;
                     }
                  }
                  
                  if(!alreadyAdded)
                  {
                     // Make symbol available for selection
                     SymbolSelect(symbolName, true);
                     m_filteredSymbols[m_filteredCount] = symbolName;
                     m_filteredCount++;
                  }
               }
            }
         }
      }
   }
   
   m_browserOffset = 0;
   UpdateBrowserDisplay();
   
   Print("Filter result v56: ", m_filteredCount, " symbols");
}

//+------------------------------------------------------------------+
//| DROPDOWN EVENT HANDLING - FIXED CLICK DETECTION               |
//+------------------------------------------------------------------+
bool CTradingAssistantFixed::HandleDropdownEvents(const int id, const long &lparam, const double &dparam, const string &sparam)
{
   if(!m_dropdownVisible) return false;
   
   if(id == CHARTEVENT_OBJECT_CLICK)
   {
      // === DROPDOWN CONTROLS v56 - FIXED BUTTON DETECTION ===
      if(sparam == "DropdownClose_v56" || sparam == "DropdownClose_v56_Text")
      {
         Print("Dropdown close clicked v56");
         DestroyDropdownObjects();
         m_dropdownVisible = false;
         ChartRedraw(0);
         return true;
      }
      
      if(sparam == "DropdownScrollUp_v56" || sparam == "DropdownScrollUp_v56_Text")
      {
         Print("Dropdown scroll up v56");
         if(m_dropdownOffset > 0)
         {
            m_dropdownOffset -= 3;
            if(m_dropdownOffset < 0) m_dropdownOffset = 0;
            UpdateDropdownDisplay();
         }
         return true;
      }
      
      if(sparam == "DropdownScrollDown_v56" || sparam == "DropdownScrollDown_v56_Text")
      {
         Print("Dropdown scroll down v56");
         m_dropdownOffset += 3;
         UpdateDropdownDisplay();
         return true;
      }
      
      // === FIXED DROPDOWN SYMBOL SELECTION v56 ===
      if(StringFind(sparam, "DropdownSymbol_v56_") == 0)
      {
         Print("FIXED: Dropdown symbol clicked: ", sparam);
         
         // Handle both rectangle and text object clicks
         string baseObjName = sparam;
         if(StringFind(sparam, "_Text") >= 0)
         {
            baseObjName = StringSubstr(sparam, 0, StringFind(sparam, "_Text"));
         }
         
         // Extract button index
         string indexStr = StringSubstr(baseObjName, StringLen("DropdownSymbol_v56_"));
         int buttonIndex = (int)StringToInteger(indexStr);
         
         Print("FIXED: Button index: ", buttonIndex);
         
         if(buttonIndex >= 0 && buttonIndex < 8)
         {
            string textObjName = "DropdownSymbol_v56_" + IntegerToString(buttonIndex) + "_Text";
            string selectedSymbol = ObjectGetString(0, textObjName, OBJPROP_TEXT);
            
            Print("FIXED: Selected symbol: '", selectedSymbol, "'");
            
            if(StringLen(selectedSymbol) > 0)
            {
               Print("✅ FIXED: Dropdown symbol selected v56: ", selectedSymbol);
               ChangeCurrentSymbol(selectedSymbol);
               
               // Auto-close dropdown
               DestroyDropdownObjects();
               m_dropdownVisible = false;
               ChartRedraw(0);
            }
         }
         return true;
      }
   }
   
   // === TEXT FIELD CHANGES v56 ===
   if(id == CHARTEVENT_OBJECT_ENDEDIT)
   {
      if(sparam == "DropdownSearch_v56")
      {
         Print("Dropdown search changed v56");
         m_dropdownOffset = 0;
         UpdateDropdownDisplay();
         return true;
      }
   }
   
   return false;
}

            ObjectSetInteger(0, removeObjName + "_Text", OBJPROP_COLOR, C'255,255,255');
         }
         else
         {
            ObjectSetInteger(0, removeObjName, OBJPROP_BGCOLOR, C'128,128,128'); // Gray (nothing to remove)
            ObjectSetString(0, removeObjName + "_Text", OBJPROP_TEXT, "○");
            ObjectSetInteger(0, removeObjName + "_Text", OBJPROP_COLOR, C'0,0,0');
         }
      }
      else
      {
         // FIXED: Hide unused buttons properly - NO "Label" text
         ObjectSetString(0, symbolObjName + "_Text", OBJPROP_TEXT, "");
         ObjectSetString(0, addObjName + "_Text", OBJPROP_TEXT, "");
         ObjectSetString(0, removeObjName + "_Text", OBJPROP_TEXT, "");
         ObjectSetInteger(0, symbolObjName, OBJPROP_BGCOLOR, C'240,240,240'); // Light gray
         ObjectSetInteger(0, addObjName, OBJPROP_BGCOLOR, C'240,240,240');
         ObjectSetInteger(0, removeObjName, OBJPROP_BGCOLOR, C'240,240,240');
      }
   }
   
   Print("Browser showing ", m_filteredCount, " symbols (offset: ", m_browserOffset, ")");
   ChartRedraw(0);
}

//+------------------------------------------------------------------+
//| BROWSER EVENT HANDLING - FIXED CLICK DETECTION                |
//+------------------------------------------------------------------+
bool CTradingAssistantFixed::HandleBrowserEvents(const int id, const long &lparam, const double &dparam, const string &sparam)
{
   if(!m_browserVisible) return false;
   
   if(id == CHARTEVENT_OBJECT_CLICK)
   {
      // === BROWSER CONTROLS v56 - FIXED BUTTON DETECTION ===
      if(sparam == "BrowserClose_v56" || sparam == "BrowserClose_v56_Text")
      {
         Print("Browser close clicked v56");
         DestroyBrowserObjects();
         m_browserVisible = false;
         ChartRedraw(0);
         return true;
      }
      
      if(sparam == "BrowserScrollUp_v56" || sparam == "BrowserScrollUp_v56_Text")
      {
         Print("Browser scroll up v56");
         if(m_browserOffset > 0)
         {
            m_browserOffset -= 4;
            if(m_browserOffset < 0) m_browserOffset = 0;
            UpdateBrowserDisplay();
         }
         return true;
      }
      
      if(sparam == "BrowserScrollDown_v56" || sparam == "BrowserScrollDown_v56_Text")
      {
         Print("Browser scroll down v56");
         m_browserOffset += 4;
         if(m_browserOffset >= m_filteredCount) m_browserOffset = m_filteredCount - 12;
         if(m_browserOffset < 0) m_browserOffset = 0;
         UpdateBrowserDisplay();
         return true;
      }
      
      // === FIXED BROWSER SYMBOL CLICKS v56 ===
      if(StringFind(sparam, "BrowserSymbol_v56_") == 0)
      {
         string indexStr = StringSubstr(sparam, StringLen("BrowserSymbol_v56_"));
         int buttonIndex = (int)StringToInteger(indexStr);
         
         if(buttonIndex >= 0 && buttonIndex < 12)
         {
            string textObjName = "BrowserSymbol_v56_" + IntegerToString(buttonIndex) + "_Text";
            string selectedSymbol = ObjectGetString(0, textObjName, OBJPROP_TEXT);
            if(StringLen(selectedSymbol) > 0)
            {
               Print("FIXED: Browser symbol selected v56: ", selectedSymbol);
               ChangeCurrentSymbol(selectedSymbol);
            }
         }
         return true;
      }
      
      // === FIXED BROWSER ADD/REMOVE BUTTONS v56 ===
      if(StringFind(sparam, "BrowserAdd_v56_") == 0)
      {
         string indexStr = StringSubstr(sparam, StringLen("BrowserAdd_v56_"));
         int buttonIndex = (int)StringToInteger(indexStr);
         
         if(buttonIndex >= 0 && buttonIndex < 12)
         {
            string textObjName = "BrowserSymbol_v56_" + IntegerToString(buttonIndex) + "_Text";
            string symbol = ObjectGetString(0, textObjName, OBJPROP_TEXT);
            if(StringLen(symbol) > 0)
            {
               Print("FIXED: Adding symbol to watchlist v56: ", symbol);
               AddSymbolToWatchlist(symbol);
               UpdateBrowserDisplay();
               if(m_dropdownVisible) UpdateDropdownDisplay();
            }
         }
         return true;
      }
      
      if(StringFind(sparam, "BrowserRemove_v56_") == 0)
      {
         string indexStr = StringSubstr(sparam, StringLen("BrowserRemove_v56_"));
         int buttonIndex = (int)StringToInteger(indexStr);
         
         if(buttonIndex >= 0 && buttonIndex < 12)
         {
            string textObjName = "BrowserSymbol_v56_" + IntegerToString(buttonIndex) + "_Text";
            string symbol = ObjectGetString(0, textObjName, OBJPROP_TEXT);
            if(StringLen(symbol) > 0)
            {
               Print("FIXED: Removing symbol from watchlist v56: ", symbol);
               RemoveSymbolFromWatchlist(symbol);
               UpdateBrowserDisplay();
               if(m_dropdownVisible) UpdateDropdownDisplay();
            }
         }
         return true;
      }
   }
   
   // === TEXT FIELD CHANGES v56 ===
   if(id == CHARTEVENT_OBJECT_ENDEDIT)
   {
      if(sparam == "BrowserSearch_v56")
      {
         Print("Browser search changed v56");
         string searchText = ObjectGetString(0, "BrowserSearch_v56", OBJPROP_TEXT);
         FilterBrowserSymbols(searchText);
         return true;
      }
   }
   
   return false;
}
